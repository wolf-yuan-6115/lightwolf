cmake_minimum_required(VERSION 3.13)
project(pumper_fixed_benchmarks C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(bench_reference reference.c fixtures.c ../src/eq_config.c)
target_include_directories(bench_reference PUBLIC . ../src)
target_compile_options(bench_reference PRIVATE -Wall -Wextra -Werror)
target_link_libraries(bench_reference PUBLIC m)
add_executable(test_reference test_reference.c)
target_link_libraries(test_reference PRIVATE bench_reference)
target_compile_options(test_reference PRIVATE -UNDEBUG)
add_test(NAME reference COMMAND test_reference)
add_library(bench_fixed fixed.c)
target_link_libraries(bench_fixed PUBLIC bench_reference)
target_compile_options(bench_fixed PRIVATE -Wall -Wextra -Werror)
add_executable(test_fixed test_fixed.c)
target_link_libraries(test_fixed PRIVATE bench_fixed)
target_compile_options(test_fixed PRIVATE -UNDEBUG)
add_test(NAME fixed_arithmetic COMMAND test_fixed)
add_library(bench_production_float ../src/eq_dsp.c)
target_include_directories(bench_production_float PRIVATE ../src)
target_compile_definitions(bench_production_float PRIVATE audio_controls_process=bench_no_controls)
add_executable(run_numeric run_numeric.c)
target_link_libraries(run_numeric PRIVATE bench_fixed bench_production_float)
target_compile_options(run_numeric PRIVATE -Wall -Wextra -Werror)
